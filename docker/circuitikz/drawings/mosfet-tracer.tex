% Curve tracer (mosfet-tracer): the set-up a person would wire, semantically the deck's.
%   VGK1 gk1 0 2.2   the gate of a copy, at its step                    -> source V_GS (set)
%   VK1 sw dk1 0     a 0 V source from the swept node to its drain      -> ammeter I_D (measured)
%   MK1 dk1 gk1 0    the copy, source grounded                          -> nmos
%   VSW sw 0, .dc    the swept drain supply, shared by every copy       -> source V_DS (set)
%   Sixteen copies run in one deck (eleven universal steps, five auto); the drawing shows one.
\documentclass[tikz, border=10pt]{standalone}
\usepackage{bench}
\begin{document}
\begin{tikzpicture}
\benchtitle{0,7.0}{Curve tracer}
\def\gnd{1.0}
% the transistor
\node[nmos] (Q) at (4,3) {};
\node[anchor=west, font=\small, align=left] at ($(Q.D)+(0.15,-0.55)$) {DUT};
% source to ground
\draw (Q.S) -- (Q.S |- 0,\gnd) node[ground]{};
\nodename{anchor=west}{$(Q.S)+(0.1,-0.3)$}{0}
% the ammeter in the drain, up to the rail
\coordinate (top) at ($(Q.D)+(0,1.7)$);
\ammeter{Q.D}{top}{$I_D$}
\nodename{anchor=west}{$(Q.D)+(0.1,0.25)$}{dk}
% the drain supply at the right, + up, - to ground
\coordinate (vtop) at ($(top)+(2.6,0)$);
\draw (top) -- (vtop);
\draw[set] (vtop) to[V, l={$V_{DS}$}] (vtop |- 0,\gnd) node[ground, black]{};
\nodename{anchor=south}{$(top)+(1.3,0.05)$}{sw}
% the gate: a voltage source from ground
\coordinate (g) at (Q.G);
\coordinate (gsrc) at ($(g)+(-1.4,0)$);
\draw (g) -- (gsrc);
\draw[set] (gsrc) to[V, l_={$V_{GS}$}] (gsrc |- 0,\gnd) node[ground, black]{};
\nodename{anchor=south}{$(g)+(-0.45,0.02)$}{gk}
% the definitions
\definitions{0,-0.1}{%
  $V_{GS} = 2$, 2.5, 3, 3.5, 4, 4.5, 5, 6, 8, 10 and 15 V, one copy each (the universal ladder)\\[2pt]
  and auto: the $V_{GS}$ that gave $I_D = 10$, 30, 100, 300 mA and 1 A at 15 V in the transfer sweep, the model's own\\[2pt]
  $V_{DS}$ swept 0 to 10 V in 20 mV steps;\quad read $I_D(V_{DS})$ for each step;\quad $r_{DS}$ as the chord at 0.1 V%ENDDEFS
}
\end{tikzpicture}
\end{document}
