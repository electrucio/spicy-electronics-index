% Anode characteristics (triode-anode): the set-up a person would wire, semantically the deck's.
%   VAk sw ak 0       a 0 V source from the swept node to the anode of copy k   -> ammeter I_a (measured)
%   VGk gk 0 <vg>     the grid of copy k, held                                  -> source V_g1 (set), ammeter I_g (measured)
%   XTk ak gk 0 DUT   copy k, cathode grounded                                  -> triode
%   VSW sw 0, .dc     the swept anode supply, shared by every copy              -> source V_a (set)
%   The nine copies (one per step of the model's own family) are drawn as one triode with a stepped grid.
\documentclass[tikz, border=10pt]{standalone}
\usepackage{bench}
\begin{document}
\begin{tikzpicture}
\benchtitle{0,7.0}{Anode characteristics}
\def\gnd{1.0}
% the triode
\node[triode, scale=1.35] (T) at (4,3.0) {};
\node[anchor=west, font=\small, align=left] at ($(T.anode)!0.62!(T.cathode)+(0.75,0)$) {DUT};
% the cathode to ground
\draw (T.cathode) -- (T.cathode |- 0,\gnd) node[ground]{};
\nodename{anchor=west}{$(T.cathode)+(0.1,-0.35)$}{0}
% the ammeter in the anode, up to the rail
\coordinate (top) at ($(T.anode)+(0,1.6)$);
\ammeter{T.anode}{top}{$I_a$}
\nodename{anchor=west}{$(T.anode)+(0.1,0.25)$}{ak}
% the anode supply at the right, + up, - to ground
\coordinate (vtop) at ($(top)+(2.6,0)$);
\draw (top) -- (vtop);
\draw[set] (vtop) to[V, l={$V_a$}] (vtop |- 0,\gnd) node[ground, black]{};
\nodename{anchor=south}{$(top)+(1.3,0.05)$}{sw}
% the gate: a voltage source from ground, through an ammeter
\coordinate (g) at (T.control);
\coordinate (gam) at ($(g)+(-1.7,0)$);
\ammeterh{gam}{g}{$I_g$}
\coordinate (gsrc) at ($(gam)+(-1.4,0)$);
\draw (gam) -- (gsrc);
\draw[set] (gsrc) to[V, l_={$V_{g1}$}] (gsrc |- 0,\gnd) node[ground, black]{};
\nodename{anchor=south}{$(gam)+(-0.4,0.02)$}{gk}
% the definitions
\definitions{0,-0.1}{%
  $V_{g1} = -k \cdot$ step, $k = 0, \dots, 8$: nine traces, one copy each;\quad the step the model's own, its cut-off at 250 V ($I_a = 10$ \textmu A) over 8, on the 1-2-2.5-5 series (0.5 V for a 12AX7)\\[2pt]
  $V_a$ swept 0 to 600 V in 1 V steps;\quad read $I_a(V_a)$ per trace, drawn up to where $V_a I_a$ passes $250\ \mathrm{V} \times I_0$, the model's own zero-bias power\\[2pt]
  the whole ladder (0 to $-300$ V, and the positive-grid traces) is the specification's; the catalogue runs the model's own family%ENDDEFS
}
\end{tikzpicture}
\end{document}
