% CMRR against frequency (opamp-cmrr-f): the set-up a person would wire, semantically the deck's.
%   VCM0 vcm0 0 DC 0 AC 1              the common-mode source                      -> source V_CM + v_cm (set)
%   R1_0 vcm0 inn0, R2_0 inn0 out0     the inverting side: 10 kΩ in, 10 kΩ back     -> resistors R_1, R_2
%   R3_0 vcm0 inp0, R4_0 inp0 0        the non-inverting side: 10 kΩ in, 10 kΩ down -> resistors R_3, R_4
%   XC0 inp0 inn0 vp vn out0 DUT, RL0  the difference amplifier and its load       -> op amp, resistor R_L
%   Three copies run in one deck, one per DC common mode; the drawing shows one.
\documentclass[tikz, border=10pt]{standalone}
\usepackage{bench}
\begin{document}
\begin{tikzpicture}
\benchtitle{0,8.6}{CMRR against frequency}
\def\gnd{1.0}
% the amplifier, inverting input up
\node[op amp, font=\small] (A) at (4,4.4) {DUT};
% the supply pins: the rails are what the bench sets
\rails{A.up}{A.down}{$+15$ V}{$-15$ V}
\nodename{anchor=east}{$(A.up)+(-0.08,0.3)$}{vp}
\nodename{anchor=east}{$(A.down)+(-0.08,-0.3)$}{vn}
% the output node
\draw (A.out) -- ++(0.8,0) coordinate (Ao);
\nodename{anchor=south}{$(Ao)+(-0.35,0.05)$}{out}
% the feedback, around to the inverting input
\coordinate (Afu) at ($(A.-)+(-0.5,0)$);
\coordinate (Afl) at (Afu |- 0,6.6);
\coordinate (Aol) at (Ao |- 0,6.6);
\draw (Ao) -- (Aol) to[R, l={$R_2$}] (Afl) -- (Afu) -- (A.-);
\nodename{anchor=south}{$(Afu)+(0.25,0.03)$}{inn}
% the inputs: R3 into the non-inverting one with R4 to ground, R1 into the inverting one, from the common-mode source
\draw (A.+) -- ++(-0.5,0) coordinate (pj) to[R, l=$R_3$] ++(-1.8,0) coordinate (vcm);
\draw (pj) to[R, l_=$R_4$] (pj |- 0,\gnd) node[ground]{};
\draw (vcm) -- (vcm |- A.-) to[R, l=$R_1$] (Afu);
\draw[set] (vcm) to[V, l_={$V_{CM} + v_{cm}$}] (vcm |- 0,\gnd) node[ground, black]{};
\nodename{anchor=south}{$(pj)+(0.2,0.03)$}{inp}
\nodename{anchor=south east}{$(vcm)+(-0.05,0.05)$}{vcm}
% the voltmeter on Ao
\coordinate (Avm) at ($(Ao)+(1.5,0)$);
\draw (Ao) -- (Avm);
\voltmeter{Avm |- 0,\gnd}{Avm}{$v_{out}$}
\draw (Avm |- 0,\gnd) node[ground]{};
% the load
\draw (Ao) -- ++(2.7,0) coordinate (Arl) to[R, l={$R_L$}] (Arl |- 0,\gnd) node[ground]{};
% the definitions
\definitions{0,-0.1}{%
  $R_1 = R_2 = R_3 = R_4 = 10$ k\ohm;\quad $R_L = 2$ k\ohm;\quad $V_{CM} = 0$, $-13$ and $+13$ V, one copy each;\quad $v_{cm} = 1$ V of AC\\[2pt]
  read $v_{out}(f)$, $f = 1$ Hz to 10 MHz, 20 points per decade;\quad CMRR $= v_{cm} / v_{out}$ (the resistors matched, so the amplifier's own)\\[2pt]
  at 10 Hz, 1 kHz, 10 kHz, 100 kHz and 1 MHz%ENDDEFS
}
\end{tikzpicture}
\end{document}
