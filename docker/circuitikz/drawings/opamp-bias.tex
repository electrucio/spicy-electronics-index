% Bias and offset currents (opamp-bias): the set-up a person would wire, semantically the deck's.
%   VI in 0 0, VIP in inp 0       the input at the reference, through a 0 V source   -> source V_in (set) + ammeter I_B+ (measured)
%   VNN out inn 0                 the feedback through a 0 V source                  -> ammeter I_B- (measured)
%   XB inp inn vp vn out DUT, RL  the follower and its load                          -> op amp, resistor R_L
%   The impedance decks (both inputs driven together; one driven, the other held by the servo) are separate.
\documentclass[tikz, border=10pt]{standalone}
\usepackage{bench}
\begin{document}
\begin{tikzpicture}
\benchtitle{0,7.0}{Bias and offset currents}
\def\gnd{1.0}
% the amplifier, non-inverting input up
\node[op amp, noinv input up, font=\small] (A) at (4,4.4) {DUT};
% the supply pins: the rails are what the bench sets
\rails{A.up}{A.down}{$+15$ V}{$-15$ V}
\nodename{anchor=east}{$(A.up)+(-0.08,0.3)$}{vp}
\nodename{anchor=east}{$(A.down)+(-0.08,-0.3)$}{vn}
% the output node
\draw (A.out) -- ++(0.8,0) coordinate (Ao);
\nodename{anchor=south}{$(Ao)+(-0.35,0.05)$}{out}
% the feedback, around to the inverting input
\coordinate (Afu) at ($(A.-)+(-0.5,0)$);
\coordinate (Afl) at (Afu |- 0,2.3);
\coordinate (Aol) at (Ao |- 0,2.3);
\draw (Ao) -- (Aol);
\ammeterh{Afl}{Aol}{$I_{B-}$}
\draw (Afl) -- (Afu) -- (A.-);
% the gate: a voltage source from ground, through an ammeter
\coordinate (Ai) at (A.+);
\coordinate (Aiam) at ($(Ai)+(-1.7,0)$);
\ammeterh{Aiam}{Ai}{$I_{B+}$}
\coordinate (Aisrc) at ($(Aiam)+(-1.5,0)$);
\draw (Aiam) -- (Aisrc);
\draw[set] (Aisrc) to[V, l_={$V_{in}$}] (Aisrc |- 0,\gnd) node[ground, black]{};
\nodename{anchor=south}{$(Aiam)+(-0.4,0.02)$}{in}
% the voltmeter on Ao
\coordinate (Avm) at ($(Ao)+(1.5,0)$);
\draw (Ao) -- (Avm);
\voltmeter{Avm |- 0,\gnd}{Avm}{$v_{out}$}
\draw (Avm |- 0,\gnd) node[ground]{};
% the load
\draw (Ao) -- ++(2.7,0) coordinate (Arl) to[R, l={$R_L$}] (Arl |- 0,\gnd) node[ground]{};
% the definitions
\definitions{0,-0.1}{%
  $V_{in} = 0$, or $\pm 13$ V, or through $R_S = 10$ k\ohm;\quad $R_L = 2$ k\ohm;\quad $T = -55$, 25 and 125 \textdegree C\\[2pt]
  read $I_{B+}$ and $I_{B-}$: $I_B$ their mean, $I_{OS}$ their difference;\quad $V_{OS}$ with and without $R_S$\\[2pt]
  input impedance: both inputs driven together with 1 V of AC (common mode), or one input driven and the other held (differential), 1 kHz to 1 MHz%ENDDEFS
}
\end{tikzpicture}
\end{document}
