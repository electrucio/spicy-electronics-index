% Temperature (bjt-temperature): the set-up a person would wire, semantically the deck's.
%   BT 0 bt I=…   the base drive, a log sweep               -> current source I_B (set)
%   VT ct 0 5     the collector supply; its current is IC   -> source V_CE (set) + ammeter I_C (measured)
%   QT ct bt 0    the transistor, emitter grounded          -> npn
%   .temp -55 / 100 / 125 / 150 (and 25 from the hFE deck) -> the thermometer (set)
%   The forced-ratio saturation copy (IC = 10 IB) rides in each deck; it is drawn in the saturation bench.
\documentclass[tikz, border=10pt]{standalone}
\usepackage{bench}
\begin{document}
\begin{tikzpicture}
\benchtitle{0,7.0}{Temperature}
\def\gnd{1.0}
% the transistor
\node[npn] (Q) at (4,3) {};
\node[anchor=west, font=\small, align=left] at ($(Q.C)+(0.15,-0.55)$) {DUT};
% emitter to ground
\draw (Q.E) -- (Q.E |- 0,\gnd) node[ground]{};
\nodename{anchor=west}{$(Q.E)+(0.1,-0.3)$}{0}
% the ammeter in the collector, up to the rail
\coordinate (ctop) at ($(Q.C)+(0,1.7)$);
\ammeter{Q.C}{ctop}{$I_C$}
\nodename{anchor=west}{$(Q.C)+(0.1,0.25)$}{ct}
% the supply at the right, + up, - to ground
\coordinate (vtop) at ($(ctop)+(2.6,0)$);
\draw (ctop) -- (vtop);
\draw[set] (vtop) to[V, l=$V_{CE}$] (vtop |- 0,\gnd) node[ground, black]{};
% the base: a current source from ground
\coordinate (b) at (Q.B);
\coordinate (bsrc) at ($(b)+(-1.4,0)$);
\draw (b) -- (bsrc);
\draw[set] (bsrc |- 0,\gnd) node[ground, black]{} to[I, l_=$I_B$] (bsrc);
\nodename{anchor=south}{$(b)+(-0.5,0.02)$}{bt}
% the voltmeter from the base wire to ground
\coordinate (bvm) at ($(b)+(-3.2,0)$);
\draw (b) -- (bvm);
\voltmeter{bvm |- 0,\gnd}{bvm}{$V_{BE}$}
\draw (bvm |- 0,\gnd) node[ground]{};
% the thermometer: the temperature is a simulator setting
\thermometer{$(vtop)+(1.7,-2.6)$}{$T$}
% the definitions
\definitions{0,-0.1}{%
  $I_B = 10^{k}\,\mathrm{A}$, $k = -9,\,-8.95,\,\dots,\,0$ \quad (20 points per decade, 1 nA to 1 A)\\[2pt]
  $V_{CE} = 5$ V;\quad $T = -55$, 25, 100, 125 and 150 \textdegree C, one deck each\\[2pt]
  read $h_{FE}$, $V_{BE}$, and with the saturation copy ($I_C = 10\,I_B$) $V_{CE}(sat)$, $V_{BE}(sat)$, against $I_C$ at each $T$\\[2pt]
  $dV_{BE}/dT$ against $I_C$ over each span between neighbouring temperatures%ENDDEFS
}
\end{tikzpicture}
\end{document}
