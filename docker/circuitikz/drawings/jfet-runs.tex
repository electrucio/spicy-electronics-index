% Operating point (jfet-runs): the set-up a person would wire, semantically the deck's.
%   VT1 dt1 0 10   the 10 V copy's drain supply; its current is ID   -> source V_DS (set) + ammeter I_D (measured)
%   VIG1 g gi1 0   a 0 V source from the gate drive into the gate     -> ammeter I_G (measured)
%   JT1 dt1 gi1 0  the transistor, source grounded                   -> njfet
%   read where the transfer sweep's gate passes through 0 V: IDSS, IG
\documentclass[tikz, border=10pt]{standalone}
\usepackage{bench}
\begin{document}
\begin{tikzpicture}
\benchtitle{0,7.0}{Operating point}
\def\gnd{1.0}
% the transistor
\node[njfet] (Q) at (4,3) {};
\node[anchor=west, font=\small, align=left] at ($(Q.D)+(0.15,-0.55)$) {DUT};
% source to ground
\draw (Q.S) -- (Q.S |- 0,\gnd) node[ground]{};
\nodename{anchor=west}{$(Q.S)+(0.1,-0.3)$}{0}
% the ammeter in the drain, up to the rail
\coordinate (top) at ($(Q.D)+(0,1.7)$);
\ammeter{Q.D}{top}{$I_D$}
\nodename{anchor=west}{$(Q.D)+(0.1,0.25)$}{dt1}
% the drain supply at the right, + up, - to ground
\coordinate (vtop) at ($(top)+(2.6,0)$);
\draw (top) -- (vtop);
\draw[set] (vtop) to[V, l={$V_{DS}$}] (vtop |- 0,\gnd) node[ground, black]{};
% the gate: a voltage source from ground, through an ammeter
\coordinate (g) at (Q.G);
\coordinate (gam) at ($(g)+(-1.7,0)$);
\ammeterh{gam}{g}{$I_G$}
\coordinate (gsrc) at ($(gam)+(-1.4,0)$);
\draw (gam) -- (gsrc);
\draw[set] (gsrc) to[V, l_={$V_{GS}$}] (gsrc |- 0,\gnd) node[ground, black]{};
\nodename{anchor=south}{$(gam)+(-0.4,0.02)$}{g}
% the definitions
\definitions{0,-0.1}{%
  $V_{GS} = 0$, $V_{DS} = 10$ V: the point of the transfer sweep where the gate passes through 0 V\\[2pt]
  read $I_D = I_{DSS}$ and $I_G$: does the card load, and does the solver settle?%ENDDEFS
}
\end{tikzpicture}
\end{document}
