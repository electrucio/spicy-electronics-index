% Phase reversal (opamp-phase-reversal): the set-up a person would wire, semantically the deck's.
%   VP vp 0 15, VN vn 0 -15     the rails                                          -> the supply pins (set)
%   VIN in 0 0, VIM in inp 0                  the swept source, through a 0 V source   -> source V_in (set) + ammeter I_in (measured)
%   X in out vp vn out DUT      the amplifier as a follower: in+, in-, V+, V-, out  -> op amp, output to the inverting input
%   RL out 0 10000               the load                                           -> resistor R_L
%   V(out) read                                                                    -> voltmeter v_out (measured)
\documentclass[tikz, border=10pt]{standalone}
\usepackage{bench}
\begin{document}
\begin{tikzpicture}
\benchtitle{0,7.0}{Phase reversal}
\def\gnd{1.0}
% the amplifier, non-inverting input up
\node[op amp, noinv input up, font=\small] (A) at (4,4.4) {DUT};
% the supply pins: the rails are what the bench sets
\rails{A.up}{A.down}{$+15$ V}{$-15$ V}
\nodename{anchor=east}{$(A.up)+(-0.08,0.3)$}{vp}
\nodename{anchor=east}{$(A.down)+(-0.08,-0.3)$}{vn}
% the output node
\draw (A.out) -- ++(0.8,0) coordinate (Ao);
\nodename{anchor=south}{$(Ao)+(-0.35,0.05)$}{out}
% the feedback, around to the inverting input
\coordinate (Afu) at ($(A.-)+(-0.5,0)$);
\coordinate (Afl) at (Afu |- 0,2.3);
\coordinate (Aol) at (Ao |- 0,2.3);
\draw (Ao) -- (Aol) -- (Afl) -- (Afu) -- (A.-);
% the gate: a voltage source from ground, through an ammeter
\coordinate (Ai) at (A.+);
\coordinate (Aiam) at ($(Ai)+(-1.7,0)$);
\ammeterh{Aiam}{Ai}{$I_{in}$}
\coordinate (Aisrc) at ($(Aiam)+(-1.5,0)$);
\draw (Aiam) -- (Aisrc);
\draw[set] (Aisrc) to[V, l_={$V_{in}$}] (Aisrc |- 0,\gnd) node[ground, black]{};
\nodename{anchor=south}{$(Aiam)+(-0.4,0.02)$}{in}
% the voltmeter on Ao
\coordinate (Avm) at ($(Ao)+(1.5,0)$);
\draw (Ao) -- (Avm);
\voltmeter{Avm |- 0,\gnd}{Avm}{$v_{out}$}
\draw (Avm |- 0,\gnd) node[ground]{};
% the load
\draw (Ao) -- ++(2.7,0) coordinate (Arl) to[R, l={$R_L$}] (Arl |- 0,\gnd) node[ground]{};
% the definitions
\definitions{0,-0.1}{%
  $V_{in}$ swept $-17$ to $+17$ V in 0.1 V steps, 2 V beyond each rail;\quad $R_L = 10$ k\ohm\\[2pt]
  read $v_{out}$ and $I_{in}$: whether the output reverses beyond the common-mode range, and the input current at $-16$, 14 and 16 V (the input protection)%ENDDEFS
}
\end{tikzpicture}
\end{document}
