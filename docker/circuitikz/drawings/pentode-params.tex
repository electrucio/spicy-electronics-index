% Transconductance, anode resistance and the screen's amplification factor (pentode-params): the set-up a person would wire, semantically the deck's.
%   copy A: AC on the control grid   -> g_m = i_a / v_g1
%   copy B: AC on the anode          -> 1/r_a = i_a / v_a
%   copy C: AC on the screen         -> g_m2 = i_a / v_g2;  mu(g1-g2) = g_m / g_m2
%   .ac lin 1 1k 1k; one set per (Vg2, Ia) of the ladders

\begin{document}

\def\gnd{1.0}

\node[pentode, scale=1.6] (TA) at (4,3.0) {};
\node[anchor=west, font=\small, align=left] at ($(TA.anode)!0.62!(TA.cathode)+(0.95,0)$) {A};
% the cathode to ground
\draw (TA.cathode) -- (TA.cathode |- 0,\gnd) node[ground]{};

\coordinate (Atop) at ($(TA.anode)+(0,1.6)$);
\ammeter{TA.anode}{Atop}{$I_a + i_a$}
% the anode supply at the right, + up, - to ground
\coordinate (Avtop) at ($(Atop)+(2.6,0)$);
\draw (Atop) -- (Avtop);
\draw[set] (Avtop) to[V, l={$V_a$}] (Avtop |- 0,\gnd) node[ground, black]{};

\coordinate (Ag) at (TA.control);
\coordinate (Agsrc) at ($(Ag)+(-1.1,0)$);
\draw (Ag) -- (Agsrc);
\draw[set] (Agsrc) to[V, l_={$V_{g1} + v$}] (Agsrc |- 0,\gnd) node[ground, black]{};
% the screen, out to the left above the grid's circuit, to its own supply
\draw (TA.screen) -- ++(-0.3,0) -- ++(0,1.25) -- ++(-0.8,0) coordinate (As);
