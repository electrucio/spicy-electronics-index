% Overload recovery (opamp-overload): the set-up a person would wire, semantically the deck's.
%   VIN in 0 PULSE(0 1.8 …), R1 in inn 1000   the pulse through 1 kΩ into the inverting input   -> source V_in (set) + resistor R_1
%   R2 out inn 10000                           the feedback                                    -> resistor R_2
%   XS 0 inn vp vn out DUT, RL out 0 2000      the inverter at G = -10, non-inverting input grounded   -> op amp, resistor R_L
%   V(out) read                                                                                -> voltmeter v_out (measured)
%   One deck per polarity.
\documentclass[tikz, border=10pt]{standalone}
\usepackage{bench}
\begin{document}
\begin{tikzpicture}
\benchtitle{0,8.6}{Overload recovery}
\def\gnd{1.0}
% the amplifier, inverting input up
\node[op amp, font=\small] (A) at (4,4.4) {DUT};
% the supply pins: the rails are what the bench sets
\rails{A.up}{A.down}{$+15$ V}{$-15$ V}
\nodename{anchor=east}{$(A.up)+(-0.08,0.3)$}{vp}
\nodename{anchor=east}{$(A.down)+(-0.08,-0.3)$}{vn}
% the output node
\draw (A.out) -- ++(0.8,0) coordinate (Ao);
\nodename{anchor=south}{$(Ao)+(-0.35,0.05)$}{out}
% the feedback, around to the inverting input
\coordinate (Afu) at ($(A.-)+(-0.5,0)$);
\coordinate (Afl) at (Afu |- 0,6.6);
\coordinate (Aol) at (Ao |- 0,6.6);
\draw (Ao) -- (Aol) to[R, l={$R_2$}] (Afl) -- (Afu) -- (A.-);
\nodename{anchor=south}{$(Afu)+(0.25,0.03)$}{inn}
% the inverting input: R1 from the pulse source
\draw (Afu) to[R, l=$R_1$] ++(-1.8,0) coordinate (vin);
\draw[set] (vin) to[V, l_=$V_{in}$] (vin |- 0,\gnd) node[ground, black]{};
\nodename{anchor=south east}{$(vin)+(-0.05,0.05)$}{in}
% the non-inverting input grounded
\draw (A.+) -- ++(-0.6,0) coordinate (Apg) -- (Apg |- 0,\gnd) node[ground]{};
\nodename{anchor=south}{$(A.+)+(-0.3,0.03)$}{0}
% the voltmeter on Ao
\coordinate (Avm) at ($(Ao)+(1.5,0)$);
\draw (Ao) -- (Avm);
\voltmeter{Avm |- 0,\gnd}{Avm}{$v_{out}$}
\draw (Avm |- 0,\gnd) node[ground]{};
% the load
\draw (Ao) -- ++(2.7,0) coordinate (Arl) to[R, l={$R_L$}] (Arl |- 0,\gnd) node[ground]{};
% the definitions
\definitions{0,-0.1}{%
  $R_1 = 1$ k\ohm, $R_2 = 10$ k\ohm: $G = -10$;\quad $V_{in}$: a 1.8 V pulse at 1 \textmu s, 10 ns edges, 20 \textmu s wide: 18 V demanded of the output, one deck per polarity;\quad $R_L = 2$ k\ohm\\[2pt]
  read $v_{out}(t)$: the time from the pulse's end to the output back within 100 mV of its rest%ENDDEFS
}
\end{tikzpicture}
\end{document}
