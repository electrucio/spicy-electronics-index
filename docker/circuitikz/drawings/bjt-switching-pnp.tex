% Switching times (bjt-switching): the set-up a person would wire, semantically the deck's.
%   VIN in 0 PWL(…)   the drive pulse                            -> source V_in (set)
%   RB in b           the base resistor, 100 V / IC              -> resistor R_B
%   VCC vcc 0 10      the supply, 10 or 30 V                     -> source V_CC (set)
%   RL vcc cs         the load, VCC / IC                         -> resistor R_L
%   VS cs c 0         a 0 V source in the collector: reads IC    -> ammeter I_C (measured)
%   Q1 c b 0          the transistor, emitter grounded           -> npn
%   One deck per switched current (1-2-5, 10 mA to 10 A) and supply.
\documentclass[tikz, border=10pt]{standalone}
\usepackage{bench}
\begin{document}
\begin{tikzpicture}
\benchtitle{0,8.2}{Switching times, PNP}
\def\gnd{1.0}
% the transistor
\node[pnp, yscale=-1] (Q) at (4,3) {};
\node[anchor=west, font=\small, align=left] at ($(Q.C)+(0.15,-0.55)$) {DUT};
% emitter to ground
\draw (Q.E) -- (Q.E |- 0,\gnd) node[ground]{};
\nodename{anchor=west}{$(Q.E)+(0.1,-0.3)$}{0}
% the ammeter in the collector, up to the rail
\coordinate (ctop) at ($(Q.C)+(0,1.6)$);
\ammeter{Q.C}{ctop}{$I_C(t)$}
\nodename{anchor=west}{$(Q.C)+(0.1,0.25)$}{c}
\nodename{anchor=west}{$(ctop)+(0.1,0.2)$}{cs}
% the load up to the rail
\draw (ctop) to[R, l_=$R_L$] ++(0,1.6) coordinate (rtop);
% the supply at the right, + up, - to ground
\coordinate (vtop) at ($(rtop)+(2.6,0)$);
\draw (rtop) -- (vtop);
\draw[set] (vtop) to[V, invert, l=$V_{CC}$] (vtop |- 0,\gnd) node[ground, black]{};
\nodename{anchor=south}{$(rtop)+(1.3,0.05)$}{vcc}
% the base resistor and the pulse source
\coordinate (b) at (Q.B);
\draw (b) -- ++(-0.3,0) coordinate (bj) to[R, l_=$R_B$] ++(-1.8,0) coordinate (vin);
\draw[set] (vin) to[V, invert, l_=$V_{in}$] (vin |- 0,\gnd) node[ground, black]{};
\nodename{anchor=south}{$(b)+(-0.15,0.05)$}{b}
\nodename{anchor=south east}{$(vin)+(-0.05,0.05)$}{in}
% the definitions
\definitions{0,-0.1}{%
  $V_{in}$: $-9.3$ V, then $+10.7$ V from 100 ns to 3.1 \textmu s, then $-9.3$ V;\quad 2 ns edges\\[2pt]
  $R_B = 100\ \mathrm{V} / I_C$, so $I_B = +I_C/10$ on and $-I_C/10$ off past a 0.7 V $V_{BE}$\\[2pt]
  $R_L = V_{CC} / I_C$;\quad $V_{CC} = 10$ and 30 V;\quad $I_C = 10$, 20, 50, 100 mA $\dots$ 10 A (1-2-5), one deck each\\[2pt]
  read $I_C(t)$: $t_d$, $t_r$, $t_s$, $t_f$ between the 10\,\% and 90\,\% levels\\[2pt]
  PNP: every source reversed; values written as magnitudes%ENDDEFS
}
\end{tikzpicture}
\end{document}
