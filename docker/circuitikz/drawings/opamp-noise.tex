% Noise (opamp-noise): the set-up a person would wire, semantically the deck's.
%   VP vp 0 15, VN vn 0 -15      the rails                                       -> the supply pins (set)
%   VIN in 0 DC 0 AC 1           the noise input on the non-inverting input       -> source v_in (set)
%   RS in inr <rs>               the source resistance (0 in the first deck)      -> resistor R_S
%   XN inr out vp vn out DUT     the follower                                    -> op amp, output to the inverting input
%   RL out 0 2000                the load                                        -> resistor R_L
%   .noise V(out) VIN            the output noise, referred to v_in              -> voltmeter v_out (measured)
\documentclass[tikz, border=10pt]{standalone}
\usepackage{bench}
\begin{document}
\begin{tikzpicture}
\benchtitle{0,7.0}{Noise}
\def\gnd{1.0}
% the amplifier, non-inverting input up
\node[op amp, noinv input up, font=\small] (A) at (4,4.4) {DUT};
% the supply pins: the rails are what the bench sets
\rails{A.up}{A.down}{$+15$ V}{$-15$ V}
\nodename{anchor=east}{$(A.up)+(-0.08,0.3)$}{vp}
\nodename{anchor=east}{$(A.down)+(-0.08,-0.3)$}{vn}
% the output node
\draw (A.out) -- ++(0.8,0) coordinate (Ao);
\nodename{anchor=south}{$(Ao)+(-0.35,0.05)$}{out}
% the feedback, around to the inverting input
\coordinate (Afu) at ($(A.-)+(-0.5,0)$);
\coordinate (Afl) at (Afu |- 0,2.3);
\coordinate (Aol) at (Ao |- 0,2.3);
\draw (Ao) -- (Aol) -- (Afl) -- (Afu) -- (A.-);
% the input: the source resistance and the source
\coordinate (Ai) at (A.+);
\draw (Ai) -- ++(-0.3,0) coordinate (Aij) to[R, l_={$R_S$}] ++(-1.8,0) coordinate (Aisrc);
\draw[set] (Aisrc) to[V, l_={$v_{in}$}] (Aisrc |- 0,\gnd) node[ground, black]{};
\nodename{anchor=south east}{$(Aisrc)+(-0.05,0.05)$}{in}
% the voltmeter on Ao
\coordinate (Avm) at ($(Ao)+(1.5,0)$);
\draw (Ao) -- (Avm);
\voltmeter{Avm |- 0,\gnd}{Avm}{$v_{out}$}
\draw (Avm |- 0,\gnd) node[ground]{};
% the load
\draw (Ao) -- ++(2.7,0) coordinate (Arl) to[R, l={$R_L$}] (Arl |- 0,\gnd) node[ground]{};
% the definitions
\definitions{0,-0.1}{%
  $R_S = 0$ gives $e_n$;\quad $R_S = 100$ k\ohm{} gives $e_n^2 + i_n^2 R_S^2 + 4kTR_S$, from which $i_n$;\quad $R_L = 2$ k\ohm\\[2pt]
  $f = 0.1$ Hz to 1 MHz, 20 points per decade;\quad read the noise of $v_{out}$ referred to $v_{in}$: $e_n$ and $i_n$ at 10 Hz, 100 Hz, 1 kHz and 10 kHz, the $1/f$ corner\\[2pt]
  integrated 0.1 to 10 Hz (peak to peak), 10 Hz to 10 kHz and 20 Hz to 20 kHz;\quad also at $\pm 5$ V%ENDDEFS
}
\end{tikzpicture}
\end{document}
